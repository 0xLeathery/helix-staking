    ```rust
    // === Phase 24: Boost auto-link ===
    let mut seed_balance_at_stake: u64 = 0;

    // Check if BoostRecord was passed in remaining_accounts
    // Index depends on whether ClaimConfig was also passed
    let boost_start_idx = if ctx.remaining_accounts.len() >= 3 { 1 } else { usize::MAX };

    if boost_start_idx < ctx.remaining_accounts.len() {
        let boost_record_info = &ctx.remaining_accounts[boost_start_idx];
        let seed_ata_info = &ctx.remaining_accounts[boost_start_idx + 1];

        // Validate boost record PDA
        let (expected_boost_pda, _) = Pubkey::find_program_address(
            &[BOOST_RECORD_SEED, ctx.accounts.user.key().as_ref()],
            ctx.program_id,
        );

        if boost_record_info.key() == expected_boost_pda {
            if let Ok(boost_record) = Account::<BoostRecord>::try_from(boost_record_info) {
                if boost_record.boosted_stake_id == u64::MAX
                    && global_state.get_boost_enabled()
                    && global_state.get_seed_mint() != Pubkey::default()
                {
                    // Read seed ATA balance (parse amount at offset 64, 8 bytes LE)
                    let data = seed_ata_info.try_borrow_data()?;
                    if data.len() >= 72 {
                        seed_balance_at_stake = u64::from_le_bytes(
                            data[64..72].try_into().unwrap()
                        );
                    }

                    if seed_balance_at_stake >= global_state.get_min_seed_balance() {
                        // Write boosted_stake_id back to BoostRecord
                        // Must manually serialize since remaining_accounts aren't managed by Anchor
                        let mut boost_data = boost_record_info.try_borrow_mut_data()?;
                        // boosted_stake_id is at offset: 8 (discriminator) + 32 (user) + 8 (slot) + 1 (bump) = 49
                        let stake_id_bytes = stake_account.stake_id.to_le_bytes();
                        boost_data[49..57].copy_from_slice(&stake_id_bytes);
                    } else {
                        seed_balance_at_stake = 0; // Below minimum, don't link
                    }
                }
            }
        }
    }

    stake_account.seed_balance_at_stake = seed_balance_at_stake;
    stake_account.boost_revoked = false;
    ```

    IMPORTANT: The BoostRecord write-back uses direct byte manipulation at the known offset (49 bytes in). This avoids the Anchor deserialization/re-serialization overhead and is more robust for remaining_accounts. The offset is: 8 (discriminator) + 32 (user Pubkey) + 8 (slot u64) + 1 (bump u8) = 49. The boosted_stake_id field is 8 bytes (u64 LE) at offsets 49..57.

    Also add import at top of create_stake.rs: `use crate::constants::BOOST_RECORD_SEED;` and `use crate::state::BoostRecord;`.
